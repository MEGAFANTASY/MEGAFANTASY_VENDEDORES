function doGet(e) {
  const hoja = e.parameter.hoja || 'megafantasy';
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(hoja);

  if (!sheet) {
    return ContentService
      .createTextOutput(JSON.stringify({ error: 'Hoja no encontrada: ' + hoja }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const data = sheet.getDataRange().getValues();
  const headers = data[0].map(h => String(h).toLowerCase().trim().replace(/ /g, '_'));
  const rows = [];

  for (let i = 1; i < data.length; i++) {
    const obj = {};
    for (let j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    rows.push(obj);
  }

  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

// doPost: escribir confirmacion de transportadora en la hoja
// Recibe JSON: { hoja, factura, valor }
// Busca la fila por FACTURA y escribe valor en CONFIRMADO APP VENDEDOR
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var hoja = data.hoja;
  var factura = data.factura;
  var valor = data.valor;

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(hoja);

  if (!sheet) {
    return ContentService
      .createTextOutput(JSON.stringify({ error: 'Hoja no encontrada: ' + hoja }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    return ContentService
      .createTextOutput(JSON.stringify({ error: 'Sin datos' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]
    .map(function(h) { return String(h).toLowerCase().trim().replace(/ /g, '_'); });

  var facturaCol = headers.indexOf('factura');
  var confirmadoCol = headers.indexOf('confirmado_app_vendedor');

  if (facturaCol === -1 || confirmadoCol === -1) {
    return ContentService
      .createTextOutput(JSON.stringify({ error: 'Columnas no encontradas' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var dataRange = sheet.getRange(2, facturaCol + 1, lastRow - 1, 1).getValues();

  for (var i = 0; i < dataRange.length; i++) {
    if (String(dataRange[i][0]).trim() === String(factura).trim()) {
      sheet.getRange(i + 2, confirmadoCol + 1).setValue(valor);
      return ContentService
        .createTextOutput(JSON.stringify({ ok: true, fila: i + 2 }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({ error: 'Factura no encontrada: ' + factura }))
    .setMimeType(ContentService.MimeType.JSON);
}
